import json import numpy as np from datetime import datetime class ThreatFeatureExtractor: def __init__(self): self.model = self.load_model() self.feature_cache = {} def extract_features(self, log_entry): """Extract ML features from security logs""" features = { 'timestamp': self.get_timestamp_features(log_entry['eventTime']), 'api_pattern': self.analyze_api_pattern(log_entry['eventName']), 'ip_behavior': self.calculate_ip_entropy(log_entry['sourceIPAddress']), 'temporal_context': self.get_temporal_context(log_entry), 'resource_access': self.analyze_resource_access(log_entry), 'user_agent_anomaly': self.detect_ua_anomaly(log_entry.get('userAgent', '')), 'geolocation_risk': self.geolocation_risk_score(log_entry.get('sourceIPAddress')) } # Normalize features normalized = self.normalize_features(features) return normalized def get_timestamp_features(self, event_time): """Extract temporal features from timestamp""" dt = datetime.fromisoformat(event_time.replace('Z', '+00:00')) return { 'hour': dt.hour, 'day_of_week': dt.weekday(), 'is_weekend': 1 if dt.weekday() >= 5 else 0, 'is_business_hours': 1 if 9 <= dt.hour <= 17 else 0 } def calculate_ip_entropy(self, ip_address): """Calculate entropy score for IP address behavior""" # Implementation for IP behavior analysis return self.ip_entropy_model.predict(ip_address) def predict_threat(self, features): """Predict threat using LSTM model""" prediction = self.model.predict(np.array([features])) confidence = prediction[0][0] return { 'is_threat': confidence > 0.95, 'confidence': float(confidence), 'threat_level': 'HIGH' if confidence > 0.95 else 'MEDIUM' if confidence > 0.8 else 'LOW' }

LSTM Model Definition
import tensorflow as tf from tensorflow.keras.models import Sequential from tensorflow.keras.layers import LSTM, Dense, Dropout, Bidirectional class ThreatDetectionLSTM: def __init__(self, input_shape): self.model = self.build_model(input_shape) def build_model(self, input_shape): """Build LSTM model for threat detection""" model = Sequential([ Bidirectional(LSTM(128, return_sequences=True), input_shape=input_shape), Dropout(0.3), LSTM(64, return_sequences=False), Dropout(0.3), Dense(32, activation='relu'), Dropout(0.2), Dense(16, activation='relu'), Dense(1, activation='sigmoid') ]) model.compile( optimizer=tf.keras.optimizers.Adam(learning_rate=0.001), loss='binary_crossentropy', metrics=['accuracy', tf.keras.metrics.Precision(), tf.keras.metrics.Recall()] ) return model def train(self, X_train, y_train, X_val, y_val, epochs=50): """Train the model with early stopping""" early_stopping = tf.keras.callbacks.EarlyStopping( monitor='val_loss', patience=10, restore_best_weights=True ) history = self.model.fit( X_train, y_train, validation_data=(X_val, y_val), epochs=epochs, batch_size=32, callbacks=[early_stopping], verbose=1 ) return history def predict_anomaly(self, sequence): """Predict if sequence contains threat""" prediction = self.model.predict(sequence.reshape(1, -1, sequence.shape[1])) return { 'is_anomaly': prediction[0][0] > 0.95, 'confidence': float(prediction[0][0]), 'score': float(prediction[0][0]) * 100 }
