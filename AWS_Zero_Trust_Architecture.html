# Sample Lambda Remediation Handler
import boto3
import json
from datetime import datetime

def lambda_handler(event, context):
    # Parse Security Hub finding
    finding = event['detail']['findings'][0]
    severity = finding['Severity']['Label']
    resource_type = finding['Resources'][0]['Type']
    
    response_actions = {
        'HIGH': handle_critical_threat,
        'MEDIUM': handle_suspicious_activity,
        'LOW': handle_compliance_violation
    }
    
    # Execute appropriate response
    action_func = response_actions.get(severity, log_only)
    return action_func(finding)

def handle_critical_threat(finding):
    """Automated response for critical security threats"""
    ec2 = boto3.client('ec2')
    ssm = boto3.client('ssm')
    
    # Isolate compromised instance
    instance_id = finding['Resources'][0]['Id']
    
    # 1. Remove from Auto Scaling Group
    ec2.modify_instance_attribute(
        InstanceId=instance_id,
        DisableApiTermination={'Value': False}
    )
    
    # 2. Create forensic snapshot
    response = ec2.create_snapshots(
        InstanceSpecification={
            'InstanceId': instance_id,
            'ExcludeBootVolume': False
        }
    )
    
    # 3. Execute forensic analysis script
    ssm.send_command(
        InstanceIds=[instance_id],
        DocumentName='AWS-RunShellScript',
        Parameters={'commands': ['/opt/security/forensic.sh']}
    )
    
    return {
        'statusCode': 200,
        'remediation': 'COMPROMISED_INSTANCE_ISOLATED',
        'timestamp': datetime.utcnow().isoformat()
    }
